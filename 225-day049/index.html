 n.speed=n.state==='phone'?.5+Math.random()*.6:1.1+Math.random()*.7;

 if(n.T===TYPES.granny)n.speed*=.6;

}

function setPose(n){

 const P=n.parts;

 P.rage.visible=n.state==='angry';

 P.torso.material.emissive.setHex(n.state==='angry'?0x661010:0);

 if(n.state==='shock'){P.armL.rotation.x=P.armR.rotation.x=-2.8;P.head.rotation.x=-.2;n.pose='fixed';return;}

 if(n.hasPhone&&n.state==='phone'){

  if(n.T.acc==='selfie'){P.armR.rotation.x=-1.9;P.head.rotation.x=0;}else{P.armR.rotation.x=-1.2;P.head.rotation.x=.55;}

  P.armL.rotation.x=0;n.pose='phone';return;}

 P.head.rotation.x=(n.state==='sulk'&&!n.hasPhone)?.35:0;n.pose='free';

}

const dirV=new THREE.Vector3(),awayV=new THREE.Vector3();

function updateNPC(n,dt){

 const p=n.mesh.position,pp=player.mesh.position;

 n.cool=Math.max(0,n.cool-dt);

 let sp=0,dir=null;

 if(n.state==='phone'||n.state==='walk'){

  n.timer-=dt;

  if(n.timer<=0||p.distanceTo(n.target)<.5)newTarget(n);

  if(!n.standing){dir=dirV.subVectors(n.target,p).setY(0).normalize();sp=n.speed;}

  if(n.state==='walk'){

   const d=p.distanceTo(pp);

   if(d<3){awayV.subVectors(p,pp).setY(0).normalize();dir=dir?dir.add(awayV.multiplyScalar(1.5)).normalize():awayV;sp=Math.max(sp,1.6);}

  }

 }else if(n.state==='angry'){

  n.angryTime+=dt;n.bark-=dt;

  dir=dirV.subVectors(pp,p).setY(0);const d=dir.length();dir.normalize();

  sp=n.T.speed*n.rageSpeed;

  if(n.bark<=0){sayN(n,pick(n.T.chase),'angry',2.2);n.bark=3+Math.random()*2;}

  if(d<.75&&player.stun<=0)caughtBy(n);

  else if(n.angryTime>n.T.giveUp||d>45)giveUp(n);

 }else if(n.state==='sulk'){

  n.timer-=dt;if(n.timer<=0){n.state=n.hasPhone?'phone':'walk';setPose(n);newTarget(n);}

 }

 if(dir&&sp>0){p.addScaledVector(dir,sp*dt);n.mesh.rotation.y=lerpAngle(n.mesh.rotation.y,Math.atan2(dir.x,dir.z),1-Math.exp(-10*dt));}

 p.x=THREE.MathUtils.clamp(p.x,-S.w/2+.4,S.w/2-.4);

 pushOut(p,.3);

 animate(n,sp,dt);

}

function animate(o,sp,dt){

 const P=o.parts;o.phase+=sp*dt*2.4;

 const sw=Math.sin(o.phase)*Math.min(1,sp/2)*.6;

 P.legL.rotation.x=sw;P.legR.rotation.x=-sw;

 if(o.pose==='free'){P.armL.rotation.x=-sw*.8;P.armR.rotation.x=sw*.8;}

}

function pushOut(p,r){for(const c of pillars){const dx=p.x-c.x,dz=p.z-c.z,d=Math.hypot(dx,dz),m=c.r+r;if(d<m&&d>1e-4){p.x=c.x+dx/d*m;p.z=c.z+dz/d*m;}}}

function separate(){

 for(let i=0;i<npcs.length;i++){const a=npcs[i].mesh.position;

  for(let j=i+1;j<npcs.length;j++){const b=npcs[j].mesh.position,dx=a.x-b.x,dz=a.z-b.z,d2=dx*dx+dz*dz;

   if(d2<.3&&d2>1e-6){const d=Math.sqrt(d2),m=(.55-d)/2;a.x+=dx/d*m;a.z+=dz/d*m;b.x-=dx/d*m;b.z-=dz/d*m;}}}

}

const keys={};

addEventListener('keydown',e=>{keys[e.code]=true;if(e.code==='Space'){e.preventDefault();if(!e.repeat)onSpace();}});

const moveV=new THREE.Vector3();

function updatePlayer(dt){

 const P=player,pos=P.mesh.position;

 P.stun=Math.max(0,P.stun-dt);P.slow=Math.max(0,P.slow-dt);

 const ix=(keys.KeyD||keys.ArrowRight?1:0)-(keys.KeyA||keys.ArrowLeft?1:0);

 const iz=(keys.KeyS||keys.ArrowDown?1:0)-(keys.KeyW||keys.ArrowUp?1:0);

 moveV.set(ix,0,iz);let sp=0,dashing=false;

 if(P.stun<=0&&moveV.lengthSq()>0){

  moveV.normalize();

  dashing=(keys.ShiftLeft||keys.ShiftRight)&&P.stamina>.05;

  sp=dashing?11.5:7.5;if(P.slow>0)sp*=.5;

  P.dir.copy(moveV);pos.addScaledVector(moveV,sp*dt);

  P.mesh.rotation.y=lerpAngle(P.mesh.rotation.y,Math.atan2(moveV.x,moveV.z),1-Math.exp(-14*dt));

 }

 P.stamina=dashing?Math.max(0,P.stamina-dt*.35):Math.min(1,P.stamina+dt*.18);

 P.speedNow=sp;

 pos.x=THREE.MathUtils.clamp(pos.x,-S.w/2+.4,S.w/2-.4);pos.z=Math.min(pos.z,8);

 pushOut(pos,.3);

 player.pose='free';animate(player,sp,dt);

 P.parts.head.rotation.x=P.stun>0?.45:0;

}

function collide(){

 const pp=player.mesh.position;

 for(const n of npcs){

  const p=n.mesh.position,dx=p.x-pp.x,dz=p.z-pp.z,d=Math.hypot(dx,dz);

  if(d<.6&&d>1e-6){

   if(!event&&n.hasPhone&&n.state==='phone'&&n.cool<=0&&player.speedNow>2)startEvent(n);

   else if(n.state==='walk'&&n.cool<=0){sayN(n,'おっと…','',1.2);n.cool=2;player.slow=.4;}

   const m=.6-d;p.x+=dx/d*m*.7;p.z+=dz/d*m*.7;pp.x-=dx/d*m*.3;pp.z-=dz/d*m*.3;

  }

 }

}

function startEvent(n){

 const ph=n.phone;world.attach(ph);

 n.phone=null;n.hasPhone=false;n.state='shock';setPose(n);

 const vel=player.dir.clone().multiplyScalar(4+Math.random()*2);vel.x+=(Math.random()-.5)*3;vel.y=6+Math.random()*1.5;

 event={npc:n,phone:ph,vel,spin:new THREE.Vector3(8+Math.random()*8,Math.random()*6,6+Math.random()*8),

  real:0,c:.2+Math.random()*.6,cs:1.3+stageIndex*.15,cursor:0,resolved:false,landed:false,caught:false,broken:false};

 timeScale=.15;

 sayN(n,n.T.hit,'',2.5);

 qteText.textContent=`📱 ${n.T.short}のスマホが飛んだ！ SPACEでキャッチ！`;

 perfectEl.style.left=(event.c-.05)*100+'%';perfectEl.style.width='10%';

 grazeEl.style.left=(event.c-.15)*100+'%';grazeEl.style.width='30%';

 qteEl.style.display='block';

}

function onSpace(){

 if(state!=='play'||!event||event.resolved)return;

 const d=Math.abs(event.cursor-event.c);

 resolve(d<=.05?'perfect':d<=.15?'graze':'miss');

}

function updateEvent(rdt,gdt){

 const E=event;if(!E)return;

 if(!E.caught&&!E.landed){

  E.vel.y-=18*gdt;E.phone.position.addScaledVector(E.vel,gdt);

  E.phone.rotation.x+=E.spin.x*gdt;E.phone.rotation.y+=E.spin.y*gdt;E.phone.rotation.z+=E.spin.z*gdt;

  if(E.phone.position.y<=.03&&E.vel.y<0){

   E.phone.position.y=.03;E.landed=true;E.phone.rotation.set(-Math.PI/2,0,Math.random()*6);

   if(E.resolved)finishOutcome(E,'miss');

  }

 }

 if(!E.resolved){

  E.real+=rdt;const p=(E.real*E.cs)%2;E.cursor=p<1?p:2-p;

  cursorEl.style.left=`calc(${E.cursor*100}% - 2px)`;

  if(E.real>1.8)resolve('miss');

 }

}

function resolve(r){

 const E=event;E.resolved=true;qteEl.style.display='none';timeScale=1;

 if(r!=='miss'){

  E.caught=true;E.broken=r==='graze'&&Math.random()<.25;

  say(player.mesh,r==='perfect'?'✨キャッチ！すみません！':'あぶなっ…！','sys',1.6);

  player.stun=r==='perfect'?1.0:.6;stats.catches++;

  finishOutcome(E,r);

 }else{

  say(player.mesh,'あっ…','',1);

  if(E.landed)finishOutcome(E,'miss');

 }

}

function giveBack(n,ph){const u=ph.userData;n.mesh.add(ph);ph.position.copy(u.p);ph.rotation.copy(u.r);ph.scale.set(1,1,1);n.phone=ph;n.hasPhone=true;setPose(n);}

function finishOutcome(E,r){

 const n=E.npc,ph=E.phone,pp=player.mesh.position;

 if(r==='miss')E.broken=Math.random()<.5;

 if(E.broken){

  if(r!=='miss'){ph.position.set(pp.x+.3,.03,pp.z-.4);ph.rotation.set(-Math.PI/2,0,1);}

  ph.userData.scr.material.color.setHex(0x151515);

  stats.broken++;stats.damage+=n.T.repair;

  spawnShards(ph.position);say(ph.position.clone(),'パリーン💥','sys',1.6);flash('#fff');

  later(.7,()=>sayN(n,n.T.broken,'angry',2.6));

  log(`💥 ${n.T.name}のスマホ破損（修理代 ¥${n.T.repair.toLocaleString()}）`);

 }else{

  if(r==='miss'){say(ph.position.clone(),'セーフ✨','sys',1.4);later(1.0,()=>giveBack(n,ph));}else giveBack(n,ph);

  later(.5,()=>sayN(n,r==='perfect'?n.T.thanks:n.T.safe,'',2.4));

  log(r==='perfect'?`🙌 ${n.T.short}のスマホをキャッチ！`:`😮‍💨 ${n.T.short}のスマホ無事`);

 }

 let ch=E.broken?n.T.chaseBroken:n.T.chaseSafe;

 if(r==='perfect')ch=0;

 if(r==='graze'&&!E.broken)ch*=.5;

 n.state='sulk';n.cool=4;

 if(Math.random()<ch){n.timer=99;setPose(n);later(1.3,()=>startChase(n));}

 else{n.timer=2.5;setPose(n);}

 event=null;

}

}
